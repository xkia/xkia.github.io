🕒 time_monitor.py — 纯标准库函数耗时监控工具 ​

一个基于 Python 标准库 的轻量级函数耗时监控工具。
无需任何第三方依赖，即可实现以下功能：

  • ✅ 函数耗时监控（装饰器方式）
  • ✅ 任意代码块耗时监控（上下文方式）
  • ✅ 超时报警与自定义处理
  • ✅ 全局统计与程序退出时报告
  • ✅ 线程安全、零依赖

📦 模块实现 ​

python
"""
time_monitor.py
----------------
一个基于 Python 标准库的轻量级耗时监控工具。

功能：
1. @monitor 装饰器：用于函数耗时监控；
2. time_block 上下文管理器：用于任意代码块；
3. stats：全局统计数据；
4. 可选的阈值报警机制；
"""

import time
import functools
import threading
import atexit
from contextlib import contextmanager

# 全局耗时统计
_stats_lock = threading.Lock()
_stats = {}  # {func_name: [times]}

def _record(name, elapsed):
    """记录执行时间"""
    with _stats_lock:
        _stats.setdefault(name, []).append(elapsed)

def _report():
    """程序结束时打印简单统计"""
    if not _stats:
        return
    print("\n📊 [TimeMonitor] 函数耗时统计：")
    for name, times in _stats.items():
        avg = sum(times) / len(times)
        print(f" - {name:<20} 调用次数: {len(times):<3} 平均耗时: {avg:.3f}s")

atexit.register(_report)


# ---------------- 装饰器方式 ----------------
def monitor(threshold=None, on_exceed=None):
    """
    装饰器：监控函数执行耗时
    threshold: float，可选，超时时间（秒）
    on_exceed: callable(name, elapsed)，可选，超时回调
    """
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            start = time.perf_counter()
            result = func(*args, **kwargs)
            elapsed = time.perf_counter() - start
            _record(func.__name__, elapsed)

            if threshold and elapsed > threshold:
                msg = f"⚠️ 函数 '{func.__name__}' 超时 {elapsed:.3f}s > {threshold}s"
                print(msg)
                if on_exceed:
                    on_exceed(func.__name__, elapsed)
            return result
        return wrapper
    return decorator


# ---------------- 上下文管理器方式 ----------------
@contextmanager
def time_block(name="代码块", threshold=None, on_exceed=None):
    """
    用于监控任意代码块：
    with time_block("数据库查询", threshold=1.0):
        ...
    """
    start = time.perf_counter()
    try:
        yield
    finally:
        elapsed = time.perf_counter() - start
        _record(name, elapsed)
        print(f"⏱️ [{name}] 耗时: {elapsed:.3f}s")
        if threshold and elapsed > threshold:
            print(f"⚠️ [{name}] 超时: {elapsed:.3f}s > {threshold}s")
            if on_exceed:
                on_exceed(name, elapsed)


# ---------------- 统计接口 ----------------
def stats():
    """返回全局耗时统计结果"""
    with _stats_lock:
        return {
            name: {
                "count": len(times),
                "avg": sum(times) / len(times),
                "max": max(times),
                "min": min(times),
            }
            for name, times in _stats.items()
        }

🧠 使用示例 ​

python
from time_monitor import monitor, time_block, stats
import time, os, signal

@monitor(threshold=2)
def slow_func():
    time.sleep(2.5)

def kill_process(name, elapsed):
    print(f"🚨 {name} 超时 {elapsed:.2f}s，杀掉进程！")
    os.kill(os.getpid(), signal.SIGTERM)

@monitor(threshold=1, on_exceed=kill_process)
def risky_task():
    time.sleep(1.5)

with time_block("数据加载", threshold=1.0):
    time.sleep(1.3)

slow_func()
risky_task()

print("\n📈 当前统计：", stats())
